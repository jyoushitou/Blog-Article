# file
```cpp
#include <iostream>
#include <fstream>
#include <sstream>
#include <string>

std::string read_sql_file(const std::string& filepath)
{
    std::ifstream fin(filepath);
    if (!fin.is_open())
    {
        std::cerr << "打开文件失败：" << filepath << '\n';
        return "";
    }
    std::stringstream buf;
    buf << fin.rdbuf();
    return buf.str();
}
```
- 或者使用这个(这个对于大文件的读取快)
```
#include <iostream>
#include <fstream>
#include <string>

std::string read_sql_file(const std::string& filepath)
{
    std::ifstream fin(filepath, std::ios::ate);
    if (!fin.is_open())
    {
        std::cerr << "打开文件失败：" << filepath << '\n';
        return "";
    }
    auto len = fin.tellg();
    fin.seekg(0, std::ios::beg);

    std::string sql(static_cast<size_t>(len), '\0');
    fin.read(sql.data(), len);
    return sql;
}
```
